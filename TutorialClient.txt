-- ReplicatedFirst/TutorialClient (LocalScript) - CLEAN VERSION

local Players = game:GetService("Players")
local ReplicatedStorage = game:GetService("ReplicatedStorage")
local TweenService = game:GetService("TweenService")

local player = Players.LocalPlayer
local playerGui = player:WaitForChild("PlayerGui")

-- CREATE the ScreenGui
local gui = Instance.new("ScreenGui")
gui.Name = "TutorialGui"
gui.ResetOnSpawn = false
gui.DisplayOrder = 100
gui.Parent = playerGui

local showTutorialEvent = ReplicatedStorage:WaitForChild("ShowTutorialEvent")
local currentTutorial = nil
local autoCloseTimer = nil

-- Create tutorial popup
local function createTutorial(tutorialName, data)
	-- Remove any existing tutorial
	local existing = gui:FindFirstChild("TutorialFrame")
	if existing then existing:Destroy() end

	-- Cancel previous timer
	if autoCloseTimer then
		task.cancel(autoCloseTimer)
		autoCloseTimer = nil
	end

	currentTutorial = tutorialName

	local isOffline = (tutorialName == "OfflineEarnings")
	local frameColor = isOffline and Color3.fromRGB(255, 215, 0) or Color3.fromRGB(30, 30, 30)
	local borderColor = isOffline and Color3.fromRGB(255, 180, 0) or Color3.fromRGB(255, 200, 50)
	local frameSize = isOffline and UDim2.new(0, 550, 0, 200) or UDim2.new(0, 500, 0, 180)

	-- Create frame
	local frame = Instance.new("Frame")
	frame.Name = "TutorialFrame"
	frame.Size = frameSize
	frame.Position = UDim2.new(0.5, -frameSize.X.Offset/2, 0.5, -frameSize.Y.Offset/2)
	frame.BackgroundColor3 = frameColor
	frame.BackgroundTransparency = 0.1
	frame.BorderSizePixel = 4
	frame.BorderColor3 = borderColor
	frame.Parent = gui

	local corner = Instance.new("UICorner")
	corner.CornerRadius = UDim.new(0, 12)
	corner.Parent = frame

	local stroke = Instance.new("UIStroke")
	stroke.Color = borderColor
	stroke.Thickness = isOffline and 4 or 3
	stroke.Parent = frame

	-- ✅ FIX: Title (set ZIndex higher than frame)
	local title = Instance.new("TextLabel")
	title.Size = UDim2.new(1, -20, 0, 40)
	title.Position = UDim2.new(0, 10, 0, 10)
	title.BackgroundTransparency = 1
	title.Text = data.Title or "💡 Tip"
	title.TextColor3 = isOffline and Color3.fromRGB(0, 0, 0) or Color3.fromRGB(255, 220, 100)
	title.TextSize = isOffline and 28 or 24
	title.Font = Enum.Font.GothamBold
	title.TextXAlignment = Enum.TextXAlignment.Left
	title.ZIndex = 2  -- ✅ Higher than frame
	title.Parent = frame

	-- ✅ FIX: Message (set ZIndex higher than frame)
	local message = Instance.new("TextLabel")
	message.Size = UDim2.new(1, -20, 1, -100)
	message.Position = UDim2.new(0, 10, 0, 55)
	message.BackgroundTransparency = 1
	message.Text = data.Message or ""
	message.TextColor3 = isOffline and Color3.fromRGB(50, 50, 50) or Color3.fromRGB(255, 255, 255)
	message.TextSize = isOffline and 20 or 18
	message.Font = Enum.Font.Gotham
	message.TextWrapped = true
	message.TextXAlignment = Enum.TextXAlignment.Left
	message.TextYAlignment = Enum.TextYAlignment.Top
	message.ZIndex = 2  -- ✅ Higher than frame
	message.Parent = frame

	-- ✅ FIX: Close button (set ZIndex higher than frame)
	local closeBtn = Instance.new("TextButton")
	closeBtn.Size = UDim2.new(0, 120, 0, 40)
	closeBtn.Position = UDim2.new(0.5, -60, 1, -50)
	closeBtn.BackgroundColor3 = isOffline and Color3.fromRGB(0, 200, 0) or Color3.fromRGB(100, 200, 100)
	closeBtn.Text = "Got it! ✓"
	closeBtn.TextColor3 = Color3.fromRGB(255, 255, 255)
	closeBtn.TextSize = 18
	closeBtn.Font = Enum.Font.GothamBold
	closeBtn.ZIndex = 2  -- ✅ Higher than frame
	closeBtn.Parent = frame

	local btnCorner = Instance.new("UICorner")
	btnCorner.CornerRadius = UDim.new(0, 8)
	btnCorner.Parent = closeBtn

	-- Fade in animation
	frame.BackgroundTransparency = 1
	title.TextTransparency = 1
	message.TextTransparency = 1
	closeBtn.BackgroundTransparency = 1
	closeBtn.TextTransparency = 1
	stroke.Transparency = 1

	TweenService:Create(frame, TweenInfo.new(0.3), {BackgroundTransparency = 0.1}):Play()
	TweenService:Create(title, TweenInfo.new(0.3), {TextTransparency = 0}):Play()
	TweenService:Create(message, TweenInfo.new(0.3), {TextTransparency = 0}):Play()
	TweenService:Create(closeBtn, TweenInfo.new(0.3), {BackgroundTransparency = 0, TextTransparency = 0}):Play()
	TweenService:Create(stroke, TweenInfo.new(0.3), {Transparency = 0}):Play()

	-- Auto-dismiss
	if data.Duration then
		autoCloseTimer = task.delay(data.Duration, function()
			if frame.Parent then
				local fadeOut = TweenService:Create(frame, TweenInfo.new(0.4), {BackgroundTransparency = 1})
				fadeOut:Play()
				TweenService:Create(title, TweenInfo.new(0.4), {TextTransparency = 1}):Play()
				TweenService:Create(message, TweenInfo.new(0.4), {TextTransparency = 1}):Play()
				TweenService:Create(closeBtn, TweenInfo.new(0.4), {BackgroundTransparency = 1, TextTransparency = 1}):Play()
				TweenService:Create(stroke, TweenInfo.new(0.4), {Transparency = 1}):Play()
				fadeOut.Completed:Wait()
				frame:Destroy()
				currentTutorial = nil
				autoCloseTimer = nil
			end
		end)
	end

	-- Manual close (in TutorialClient)
	closeBtn.MouseButton1Click:Connect(function()
		print("🔵 CLIENT DEBUG: Close button clicked for " .. currentTutorial) -- ADD THIS

		local dismissEvent = ReplicatedStorage:WaitForChild("TutorialDismissedEvent")
		dismissEvent:FireServer(currentTutorial)

		if autoCloseTimer then
			task.cancel(autoCloseTimer)
			autoCloseTimer = nil
		end

		frame:Destroy()
		currentTutorial = nil
	end)
end

-- Listen for tutorials
showTutorialEvent.OnClientEvent:Connect(function(tutorialName, data)
	createTutorial(tutorialName, data)
end)

print("✅ Tutorial GUI loaded!")
